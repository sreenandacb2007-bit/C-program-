#include<stdio.h>
void main()
{
    int a[100][100],n,m,i,j,option,num,sum=0,k;
    printf("enter the number of rows");
    scanf("%d",&n);
    printf("enter the number of columns");
    scanf("%d",&m);
    printf("enter the elements of the metrix");
    for(i=0;i<n;i++)
    {
        for(j=0;j<m;j++)
        {
            scanf("%d",&a[i][j]);
        }
    }
    printf("the metrix is");
    for(i=0;i<n;i++)
    {
        printf("\n");
        for(j=0;j<m;j++)
        {
            printf("%d\t",a[i][j]);
        }
    }
    printf("\nOPTIONS\n");
    printf("1.find the sum of row\n2.find the sum of column");
    printf("\nenter your option");
    scanf("%d",&option);
    switch(option)
    {
        case 1:
        printf("enter the row number of which you want the sum");
        scanf("%d",&num);
        
        { 
            for(j=0;j<m;j++)
            {
                sum=sum+a[num-1][j];
            }
        }
        printf("sum of the row %d is %d",num,sum);
        break;
        case 2:
        printf("enter the column number which you want the sum");
        scanf("%d",&num);
        
        {
            for(i=0;i<n;i++)
            {
                sum=sum+a[i][num-1];
            }
        }
        printf("sum of the column %d is %d",num,sum);
        break;
    }
}
